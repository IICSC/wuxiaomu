STM32HAL 库记录

00. 新建工程以及 GPIO 的输出配置

难度: 入门 阅读: 5 min 更新: 2024-03-20

在 CubeMX 的 SYS 选项中，DEBUG 必须选择 Serial Wire，否则下载一次后芯片可能被锁死。

GPIO 配置： 选择引脚 -> GPIO_Output -> Pull-up (视硬件而定)。

while(1) {
    HAL_GPIO_TogglePin(GPIOA, GPIO_PIN_8); // 翻转 LED
    HAL_Delay(500); // 延时 500ms
}

01. 按键输入配置

难度: 入门 阅读: 3 min

CubeMX 配置： 选择引脚 -> GPIO_Input -> Pull-up (上拉输入，按键另一端接地)。

if (HAL_GPIO_ReadPin(GPIOC, GPIO_PIN_13) == GPIO_PIN_RESET) {
    HAL_Delay(20); // 消抖
    if (HAL_GPIO_ReadPin(GPIOC, GPIO_PIN_13) == GPIO_PIN_RESET) {
        // 执行按键按下逻辑
        while(HAL_GPIO_ReadPin(GPIOC, GPIO_PIN_13) == GPIO_PIN_RESET); // 等待松手
    }
}

02. 外部中断实现

难度: 入门 阅读: 5 min

CubeMX 配置： GPIO_EXTI -> NVIC 中使能对应的 EXTI line 中断 -> 选择 Falling Edge (下降沿触发)。

在 main.c 或 gpio.c 中重写回调函数：

void HAL_GPIO_EXTI_Callback(uint16_t GPIO_Pin) {
    if (GPIO_Pin == GPIO_PIN_0) {
        HAL_GPIO_TogglePin(GPIOB, GPIO_PIN_5); // 中断触发翻转 LED
    }
}

03. 定时器中断

难度: 入门 阅读: 8 min

CubeMX 配置： Timers -> TIMx -> Internal Clock -> 配置 PSC (预分频) 和 ARR (自动重装载)。

公式：Update Event (Hz) = Timer_Clock / ((PSC + 1) * (ARR + 1))

// 启动定时器中断
HAL_TIM_Base_Start_IT(&htim2);

// 定时器回调函数
void HAL_TIM_PeriodElapsedCallback(TIM_HandleTypeDef *htim) {
    if (htim->Instance == TIM2) {
        HAL_GPIO_TogglePin(GPIOA, GPIO_PIN_1);
    }
}

04. 串口的基础收发

难度: 入门 阅读: 10 min

CubeMX 配置： Connectivity -> USART1 -> Asynchronous -> NVIC 中使能中断。

中断接收并回显：

uint8_t res;
HAL_UART_Receive_IT(&huart1, &res, 1); // 初始开启

void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart) {
    if(huart->Instance == USART1) {
        HAL_UART_Transmit(&huart1, &res, 1, 100); // 发送回去
        HAL_UART_Receive_IT(&huart1, &res, 1); // 循环开启
    }
}

05. PWM 输出控制

难度: 入门 阅读: 8 min

CubeMX 配置： Timers -> TIMx -> Channel x -> PWM Generation CHx。配置 PSC 和 ARR 决定频率，Pulse 决定占空比。

// 开启 PWM
HAL_TIM_PWM_Start(&htim3, TIM_CHANNEL_1);

// 动态修改占空比
__HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_1, 500); // Pulse = 500

10. 输入捕获 (测量频率与占空比)

难度: 中级 阅读: 15 min

CubeMX 配置： Timers -> TIMx -> Channel x -> Input Capture Direct Mode -> 选择触发边沿。

原理：通过捕获两次上升沿之间的计数值来计算信号周期。

// 输入捕获回调
void HAL_TIM_IC_CaptureCallback(TIM_HandleTypeDef *htim) {
    if (htim->Channel == HAL_TIM_ACTIVE_CHANNEL_1) {
        uint32_t capture = HAL_TIM_ReadCapturedValue(htim, TIM_CHANNEL_1);
        // 计算频率: Freq = Timer_Clock / (PSC+1) / capture
    }
}
// 启动输入捕获中断
HAL_TIM_IC_Start_IT(&htim5, TIM_CHANNEL_1);

11. 编码器模式 (读取电机转速)

难度: 中级 阅读: 12 min

CubeMX 配置： Timers -> TIMx -> Combined Channels -> Encoder Mode -> 选择 Channel 1 & Channel 2。

编码器模式自动对 A/B 相脉冲进行加减计数，CNT 寄存器即为当前位置。

// 启动编码器
HAL_TIM_Encoder_Start(&htim3, TIM_CHANNEL_ALL);

// 读取当前计数 (带符号)
int16_t encoder_count = (int16_t)__HAL_TIM_GET_COUNTER(&htim3);

// 定时清零 (防止溢出)
__HAL_TIM_SET_COUNTER(&htim3, 0);

13. 看门狗 (独立/窗口看门狗)

难度: 中级 阅读: 8 min

独立看门狗 (IWDG)： 使用内部 LSI 时钟，不受系统时钟影响，适合检测程序跑飞。

// IWDG 初始化后启动
HAL_IWDG_Start(&hiwdg);

// 在关键循环中喂狗
HAL_IWDG_Refresh(&hiwdg);

窗口看门狗 (WWDG)： 必须在特定时间窗口内喂狗，过早或过晚都会复位，适合检测时序异常。

STM32 IAP 升级

IAP 原理与 Flash 分区

难度: 高级 阅读: 15 min

IAP (In-Application Programming) 即在应用中编程。通常将 Flash 分为 BootLoader 区和 App 区：

  • BootLoader (0x0800 0000): 负责接收固件并写入 Flash。
  • App (0x0800 4000): 实际业务逻辑。

App 中断向量表偏移：

SCB->VTOR = FLASH_BASE | 0x4000; // 在 App 的 main 函数开头设置

跳转代码实现

typedef void (*iapfun)(void);
iapfun jump2app;
void iap_load_app(uint32_t appaddr) {
    // 1. 检查栈顶地址是否合法 (是否在 SRAM 范围内)
    if(((*(volatile uint32_t*)appaddr)&0x2FFE0000) == 0x20000000) {
        // 2. 获取复位中断向量 (App 起始地址 + 4)
        jump2app = (iapfun)*(volatile uint32_t*)(appaddr + 4);
        // 3. 设置主堆栈指针 (MSP)
        __set_MSP(*(volatile uint32_t*)appaddr);
        // 4. 跳转
        jump2app();
    }
}

IMX6ULL GPIO 寄存器操作

难度: 中级 阅读: 10 min

NXP IMX6ULL 的 GPIO 控制器包含 5 组 GPIO (GPIO1-GPIO5)，每组有多个引脚。

1. 关键寄存器

  • DR (Data Register): 数据寄存器，读写引脚电平
  • GDIR (GPIO Direction Register): 方向寄存器，1=输出，0=输入
  • PSR (Pad Status Register): 引脚状态寄存器，读取实际电平
  • ICR1/ICR2 (Interrupt Config Register): 中断配置寄存器
// 配置 GPIO1_IO03 为输出
*(volatile unsigned int *)0x0209C004 |= (1 << 3);  // GDIR 设置输出
// 输出高电平
*(volatile unsigned int *)0x0209C000 |= (1 << 3);  // DR 置高
// 读取引脚状态
unsigned int status = *(volatile unsigned int *)0x0209C00C & (1 << 3);

2. IOMUX 引脚复用配置

IMX6ULL 引脚复用通过 IOMUXC 模块配置，每个引脚对应一个 SW_MUX_CTL 寄存器。

// 配置 UART1_CTS_B 为 GPIO1_IO18
*(volatile unsigned int *)0x020E0068 = 0x5;  // MUX_MODE = 5 (GPIO)
// 配置电气属性 (上拉、速度等)
*(volatile unsigned int *)0x020E02F4 = 0x10B0;

Linux 驱动开发篇

LED 驱动 (基于设备树)

难度: 中级 阅读: 12 min

通过设备树 (Device Tree) 描述硬件资源，驱动程序通过 of_ 系列函数获取资源。

1. 设备树节点 (dts)：

led {
    compatible = "alientek,led";
    status = "okay";
    led-gpio = <&gpio1 3 GPIO_ACTIVE_LOW>;
};

2. 驱动核心代码：

// 获取设备树节点
struct device_node *node = of_find_node_by_path("/led");
// 获取 GPIO 编号
int led_gpio = of_get_named_gpio(node, "led-gpio", 0);
// 申请 GPIO
gpio_request(led_gpio, "led");
// 设置为输出模式并默认拉高
gpio_direction_output(led_gpio, 1);
// 控制电平
gpio_set_value(led_gpio, 0); // 点亮 LED

蜂鸣器驱动

难度: 中级 阅读: 8 min

与 LED 类似，本质上是控制 GPIO 的高低电平。核心在于 file_operations 结构体的实现。

static ssize_t beep_write(struct file *filp, const char __user *buf, size_t cnt, loff_t *offt) {
    int ret;
    unsigned char databuf[1];
    ret = copy_from_user(databuf, buf, cnt);
    if(databuf[0] == 1) gpio_set_value(beep_gpio, 0); // 响

    return 0;
}

Linux 并发与竞争

难度: 高级 阅读: 20 min

在多线程或中断环境中，为了保护共享资源，需要使用原子操作、自旋锁或信号量。

1. 原子操作 (atomic_t)

atomic_t lock = ATOMIC_INIT(1); // 初始化原子变量为 1
if (!atomic_dec_and_test(&lock)) { // 尝试获取锁
    atomic_inc(&lock); // 恢复
    return -EBUSY;
}

2. 自旋锁 (spinlock_t)

适用于短时间持有锁的情况，锁被占用时线程会原地"自旋"等待，不进入睡眠。

spinlock_t lock;
spin_lock_init(&lock);
spin_lock_irqsave(&lock, flags); // 保护临界区，同时关闭中断
// ... 临界区代码 ...
spin_unlock_irqrestore(&lock, flags);

3. 信号量 (semaphore)

适用于长时间持有锁的场景，线程获取不到锁时进入睡眠，不占用 CPU 资源。

struct semaphore sem;
sema_init(&sem, 1); // 初始化信号量为 1 (互斥锁)
down(&sem);         // 获取信号量 (可能睡眠)
// ... 临界区代码 ...
up(&sem);           // 释放信号量

4. 互斥体 (mutex)

比信号量更严格，只能被持有一次，不允许递归锁定。

struct mutex lock;
mutex_init(&lock);
mutex_lock(&lock);    // 获取锁
// ... 临界区代码 ...
mutex_unlock(&lock);  // 释放锁

有限状态机 (FSM) 设计

难度: 中级 阅读: 15 min

状态机是 FPGA 设计的核心思想，通常采用三段式写法：状态转移、次态逻辑、输出逻辑。

// 状态定义
parameter IDLE  = 2'b00;
parameter START = 2'b01;
parameter DONE  = 2'b10;

reg [1:0] current_state, next_state;

// 第一段：状态寄存器更新
always@(posedge clk or negedge rst_n) begin
    if(!rst_n) current_state <= IDLE;
    else current_state <= next_state;
end

// 第二段：次态组合逻辑
always@(*) begin
    case(current_state)
        IDLE:  if(start_sig) next_state = START;
               else next_state = IDLE;
        START: if(done_sig) next_state = DONE;
               else next_state = START;
        DONE:  next_state = IDLE;
        default: next_state = IDLE;
    endcase
end

// 第三段：输出逻辑
always@(posedge clk or negedge rst_n) begin
    if(!rst_n) out_sig <= 0;
    else if(current_state == START) out_sig <= 1;
    else out_sig <= 0;
end

Verilog 实现 PWM 呼吸灯

难度: 中级 阅读: 10 min

通过计数器产生三角波作为占空比参考值，与固定频率的 PWM 计数器比较实现渐变效果。

module pwm_breath(
    input clk,
    input rst_n,
    output reg led
);
    reg [15:0] cnt_pwm;    // PWM 计数器
    reg [15:0] duty_cycle; // 当前占空比
    
    // PWM 计数
    always@(posedge clk) cnt_pwm <= cnt_pwm + 1;
    
    // 占空比渐变 (三角波)
    reg up_down;
    always@(posedge clk) begin
        if(cnt_pwm == 16'hFFFF) begin
            if(up_down) duty_cycle <= duty_cycle + 100;
            else duty_cycle <= duty_cycle - 100;
            if(duty_cycle >= 16'hFFFF) up_down <= 0;
            if(duty_cycle == 0) up_down <= 1;
        end
    end
    
    // PWM 比较输出
    always@(posedge clk) begin
        if(cnt_pwm < duty_cycle) led <= 1;
        else led <= 0;
    end
endmodule

环形缓冲区 (Ring Buffer) 实现

难度: 中级 阅读: 12 min

环形缓冲区是串口数据接收、任务间通信等场景的核心数据结构。

typedef struct {
    uint8_t *buffer;
    uint32_t head;
    uint32_t tail;
    uint32_t size;
} ring_buf_t;

int ring_buf_push(ring_buf_t *rb, uint8_t data) {
    uint32_t next = (rb->head + 1) % rb->size;
    if (next == rb->tail) return -1; // 满
    rb->buffer[rb->head] = data;
    rb->head = next;
    return 0;
}

int ring_buf_pop(ring_buf_t *rb, uint8_t *data) {
    if (rb->head == rb->tail) return -1; // 空
    *data = rb->buffer[rb->tail];
    rb->tail = (rb->tail + 1) % rb->size;
    return 0;
}
